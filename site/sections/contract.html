    09Coalescingopt-in coalesce('microtask' | 'frame') — sugar, not semantics
    const q = () => [[{ n: 1 }], [{ n: 2 }]], commits = { sync: 0, co: 0 }  // two chunks that settle in one tick
const feed = (c = q()) => ({ [Symbol.asyncIterator]: () => ({ next: () => Promise.resolve({ done: !c.length, value: c.shift() }) }) })
const sync = fromAsync(runtime(), feed()), co = fromAsync(runtime(), feed(), { coalesce: 'microtask' })
handleFor(sync.source).sink({ apply: () => commits.sync++ }); handleFor(co.source).sink({ apply: () => commits.co++ })
while (sync.status() === 'pending' || co.status() === 'pending') await new Promise(r => setTimeout(r))
print(`default: ${q().length} chunks → ${commits.sync} commits · coalesce('microtask'): ${commits.co} commit · same ${co.source.rowCount()} rows each`)
